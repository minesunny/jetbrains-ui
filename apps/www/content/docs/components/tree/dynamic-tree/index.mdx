## Overview

`DynamicTree` is a presentational wrapper for `@headless-tree`. You provide `containerProps` and `items` from your own tree setup, and the component renders them with the JetBrains UI tree styles.

## Usage

```tsx
import * as React from 'react';
import {
  hotkeysCoreFeature,
  selectionFeature,
  type TreeDataLoader,
} from '@headless-tree/core';
import { useTree } from '@headless-tree/react';
import {
  type DynamicTreeItemProps,
  DynamicTree,
  asyncDataLoaderFeature,
  type DynamicTreeItemData,
} from '@/components/jetbrains-ui/tree/dynamic-tree';

type FileTreeItem = DynamicTreeItemData & {
  path: string;
  isFolder?: boolean;
};

const ROOT_ID = 'root';

const loadData = async (parentId: string | null) => {
  if (parentId === null) {
    return [
      {
        id: 'folder-src',
        data: {
          label: 'src',
          path: '/src',
          isFolder: true,
        },
      },
      {
        id: 'readme-md',
        data: {
          label: 'README.md',
          path: '/README.md',
        },
      },
    ];
  }

  if (parentId === 'folder-src') {
    return [
      {
        id: 'index-tsx',
        data: {
          label: 'index.tsx',
          path: '/src/index.tsx',
        },
      },
    ];
  }

  return [];
};

function Example() {
  const loadedItemsRef = React.useRef<Record<string, FileTreeItem>>({});
  const FileItem = React.useCallback(
    ({ item }: DynamicTreeItemProps<FileTreeItem>) => (
      <div className="flex h-6 items-center gap-2 px-4 text-[13px]">
        <span>{item.isFolder() ? 'DIR' : 'FILE'}</span>
        <span className="truncate">
          {item.getItemData()?.label ?? item.getId()}
        </span>
      </div>
    ),
    [],
  );
  const dataLoader = React.useMemo<TreeDataLoader<FileTreeItem>>(
    () => ({
      getItem: async (itemId) => {
        const [realItemId] = itemId.split('@');

        if (realItemId === ROOT_ID) {
          return {
            label: 'root',
            isFolder: true,
          } as FileTreeItem;
        }

        return (
          loadedItemsRef.current[realItemId] ??
          ({
            label: realItemId,
            path: realItemId,
          } as FileTreeItem)
        );
      },
      getChildrenWithData: async (itemId) =>
        loadData(
          itemId.split('@')[0] === ROOT_ID ? null : itemId.split('@')[0],
        ),
    }),
    [],
  );
  const tree = useTree<FileTreeItem>({
    rootItemId: ROOT_ID,
    dataLoader,
    isItemFolder: (item) => Boolean(item.getItemData()?.isFolder),
    getItemName: (item) => item.getItemData()?.label ?? item.getId(),
    onLoadedItem: (itemId, item) => {
      loadedItemsRef.current[itemId] = item;
    },
    createLoadingItemData: () => ({}) as FileTreeItem,
    features: [asyncDataLoaderFeature, selectionFeature, hotkeysCoreFeature],
  });

  return (
    <DynamicTree<FileTreeItem>
      containerProps={
        tree.getContainerProps('Tree') as React.ComponentPropsWithoutRef<'div'>
      }
      items={tree.getItems()}
      item={FileItem}
      width="220px"
      height="160px"
    />
  );
}
```

## Context Menu

Right-clicking any tree item opens a context menu. The default menu is enabled automatically and provides four actions:

| Action         | Folders | Leaves | Implementation                      | Requires Prop |
| -------------- | ------- | ------ | ----------------------------------- | ------------- |
| Add            | Yes     | No     | Calls `addItem` prop                | `addItem`     |
| Refresh        | Yes     | Yes    | Built-in (invalidates data cache)   | -             |
| Clear Children | Yes     | No     | Built-in (clears cache + collapses) | -             |
| Delete         | Yes     | Yes    | Calls `deleteItem` prop             | `deleteItem`  |

Actions without a provided handler appear **disabled** in the menu.

### With action handlers

```tsx
<DynamicTree
  containerProps={tree.getContainerProps('Tree')}
  items={tree.getItems()}
  addItem={(item) => createChildNode(item.getId())}
  deleteItem={(item) => removeNode(item.getId())}
/>
```

### Disable context menu

```tsx
<DynamicTree contextMenu={null} ... />
```

### Custom context menu

```tsx
import {
  ContextMenuItem,
  ContextMenuSeparator,
} from '@/components/jetbrains-ui/context-menu';

<DynamicTree
  contextMenu={(item, actions) => (
    <>
      <ContextMenuItem onSelect={() => actions.refresh(item)}>
        Refresh
      </ContextMenuItem>
      <ContextMenuSeparator />
      <ContextMenuItem
        variant="destructive"
        onSelect={() => actions.delete?.(item)}
      >
        Delete
      </ContextMenuItem>
    </>
  )}
  deleteItem={(item) => removeNode(item.getId())}
/>;
```

The render function receives the `ItemInstance` and a `TreeItemActions` object with pre-wired helpers: `actions.refresh(item)`, `actions.clearChildren(item)`, `actions.add?.(item)`, and `actions.delete?.(item)`.

## Scrolling

`DynamicTree` already includes a `ScrollArea` wrapper. Use `width` and `height` to cap the visible viewport, and the tree scrolls once loaded items exceed the available space.

## API Highlights

- `DynamicTree<TItem>`: presentational layer that renders the tree from `containerProps` and `items`.
- `asyncDataLoaderFeature`: recommended built-in async feature with refreshable item-data cache.
- `item`: lets you replace the default `DynamicTreeItem` with your own item component.
- `contextMenu`: `undefined` for the built-in menu (default), `null` to disable, or a render function `(item, actions) => ReactNode` for a custom menu.
- `addItem`: callback to add a child item. Called by the default menu's Add action and available via `actions.add` in custom menus.
- `deleteItem`: callback to delete an item. Called by the default menu's Delete action and available via `actions.delete` in custom menus.
- `TreeItemActions<TItem>`: type for the actions object passed to the context menu render function.
- `containerProps` and `items`: provide these directly from your own `@headless-tree` setup.
- `width` and `height`: constrain the outer scroll area and enable scrolling when content overflows.
- Data loading, selection and keyboard behavior are defined by your own `@headless-tree` configuration.
