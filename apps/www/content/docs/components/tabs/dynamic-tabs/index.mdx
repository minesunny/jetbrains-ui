## Overview

`DynamicTabsList` renders a data-driven tab bar from an items array. Each tab has a close/pin button and a right-click context menu with built-in actions. Content rendering is consumer-controlled — the component only manages the tab bar.

## Usage

### Basic

```tsx
import { useState, useCallback, memo } from 'react';
import {
  DynamicTabsList,
  type DynamicTabItem,
} from '@/components/jetbrains-ui/tabs/dynamic-tabs';

const tabs: DynamicTabItem[] = [
  { id: 'main', label: 'main.ts', pinned: true },
  { id: 'app', label: 'app.tsx' },
  { id: 'utils', label: 'utils.ts' },
];

function Example() {
  const [items, setItems] = useState(tabs);
  const [activeId, setActiveId] = useState('main');

  const closeTabs = useCallback((ids: string[]) => {
    setItems((prev) => prev.filter((t) => !ids.includes(t.id)));
  }, []);

  const togglePin = useCallback((tabId: string) => {
    setItems((prev) =>
      prev.map((t) => (t.id === tabId ? { ...t, pinned: !t.pinned } : t)),
    );
  }, []);

  return (
    <>
      <DynamicTabsList
        items={items}
        activeTab={activeId}
        onActiveChange={setActiveId}
        closeTabs={closeTabs}
        togglePin={togglePin}
      />
      <div>
        {items.map((tab) => (
          <div
            key={tab.id}
            style={{ display: tab.id === activeId ? undefined : 'none' }}
          >
            <MemoizedContent tabId={tab.id} />
          </div>
        ))}
      </div>
    </>
  );
}
```

### Prevent Content Re-renders

Use `React.memo` on content components and `display: none` for inactive tabs. This keeps heavy content (editors, tables) mounted but hidden, preventing re-renders on tab switch:

```tsx
const MemoizedContent = memo(function Content({ tabId }: { tabId: string }) {
  return <Editor tabId={tabId} />;
});
```

### Disable Context Menu

```tsx
<DynamicTabsList items={items} contextMenu={null} />
```

### Custom Context Menu

```tsx
<DynamicTabsList
  items={items}
  contextMenu={(tab, index, actions) => (
    <>
      <ContextMenuItem onSelect={() => actions.close?.()}>
        Custom Close
      </ContextMenuItem>
    </>
  )}
/>
```

## Context Menu Actions

| Action              | Description                      | Disabled When                                     |
| ------------------- | -------------------------------- | ------------------------------------------------- |
| Pin Tab / Unpin Tab | Toggle pinned state              | No `togglePin` prop                               |
| Close               | Close the current tab            | Tab is pinned, or no `closeTabs` prop             |
| Close Others        | Close all other unpinned tabs    | No `closeTabs` prop                               |
| Close Left Tabs     | Close unpinned tabs to the left  | No unpinned tabs on left, or no `closeTabs` prop  |
| Close Right Tabs    | Close unpinned tabs to the right | No unpinned tabs on right, or no `closeTabs` prop |

All close-range actions automatically skip pinned tabs.
